##  Dynamic Map of Content (MOC)

```dataviewjs
const pages = dv.pages('"10.01 PROMPTS"').where(p => p.file.name !== "10.01 PROMPTS" && p.file.extension === "md");
const categories = {
  "AGENT-PROMPTS": pages.where(p => p.file.folder.includes("AGENT-PROMPTS")),
  "MODEL-PROMPTS": pages.where(p => p.file.folder.includes("MODEL-PROMPTS")),
  "OSINT-PROMPTS": pages.where(p => p.file.folder.includes("OSINT-PROMPTS")),
  "PROMPT-COLLECTIONS": pages.where(p => p.file.folder.includes("PROMPT-COLLECTIONS")),
  "PROMPT-HACKING": pages.where(p => p.file.folder.includes("PROMPT-HACKING")),
  "SYSTEM-PROMPTS": pages.where(p => p.file.folder.includes("SYSTEM-PROMPTS")),
  "Other Prompts": pages.where(p => !p.file.folder.includes("AGENT-PROMPTS") && !p.file.folder.includes("MODEL-PROMPTS") && !p.file.folder.includes("PROMPT-HACKING") && !p.file.folder.includes("SYSTEM-PROMPTS") && !p.file.folder.includes("OSINT-PROMPTS") && !p.file.folder.includes("PROMPT-COLLECTIONS"))
};

for (let [category, list] of Object.entries(categories)) {
  if (list.length === 0) continue;
  dv.header(3, `${category} (${list.length} notes)`);
  dv.table(
    ["Name", "Type / Tags", "Last Modified"],
    list.sort(p => p.file.mtime, "desc").slice(0, 15).map(p => [
      p.file.link,
      p.type || p.file.tags.join(", ") || "-",
      moment(p.file.mtime.toString()).format("YYYY-MM-DD HH:mm")
    ])
  );
}
```
