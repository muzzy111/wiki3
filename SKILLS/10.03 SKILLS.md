##  Dynamic [[Skills]] Map of Content (MOC)

```dataviewjs
const skills = dv.pages('"10.03 SKILLS"').where(p => p.file.name === "SKILL");
const verbs = ["performing", "implementing", "detecting", "analyzing", "building", "hunting", "exploiting", "testing", "conducting", "configuring", "securing", "deploying", "auditing", "reverse-engineering", "scanning"];

const grouped = {};
verbs.forEach(v => grouped[v] = []);
grouped["others"] = [];

skills.forEach(p => {
  const folderName = p.file.folder.split("/").pop();
  let matched = false;
  for (let v of verbs) {
    if (folderName.startsWith(v)) {
      grouped[v].push({ link: p.file.link, folder: folderName, page: p });
      matched = true;
      break;
    }
  }
  if (!matched) {
    grouped["others"].push({ link: p.file.link, folder: folderName, page: p });
  }
});

for (let [verb, list] of Object.entries(grouped)) {
  if (list.length === 0) continue;
  dv.header(3, `${verb.toUpperCase()} (${list.length} skills)`);
  dv.table(
    ["Skill", "Folder Name", "Description"],
    list.sort((a,b) => a.folder.localeCompare(b.folder)).slice(0, 12).map(item => [
      item.link,
      `\`${item.folder}\``,
      item.page.description || "-"
    ])
  );
}
```
